## 4.2 日记解析

### 输入格式

```markdown
# 2025-06-15

今天中午和朋友去吃了火锅，辣得我眼泪都出来了。
晚上散步看到夕阳，有点感动。

# 2025-06-16

今天加班。午饭随便吃了便利店饭团。
```

### 解析逻辑

```python
import re
from dataclasses import dataclass
from datetime import datetime


@dataclass
class DiaryEntry:
    date: str          # "2025-06-15"
    content: str       # 去掉标题后的正文
    source_file: str   # 来源文件名


def parse_diary_file(filepath: str, date_pattern: str) -> list[DiaryEntry]:
    """按日期标题切分单文件中的多篇日记。"""
    text = Path(filepath).read_text(encoding="utf-8")
    pattern = re.compile(date_pattern, re.MULTILINE)
    matches = list(pattern.finditer(text))

    if not matches:
        return []

    entries = []
    for i, match in enumerate(matches):
        date_str = match.group(1)
        start = match.end()
        end = matches[i + 1].start() if i + 1 < len(matches) else len(text)
        content = text[start:end].strip()
        if content:
            entries.append(DiaryEntry(
                date=date_str,
                content=content,
                source_file=Path(filepath).name,
            ))
    return entries
```

## 4.3 切块策略

### 为什么日记需要特殊切块

| 策略 | 适用 | 日记的问题 |
|------|------|-----------|
| 固定 500 字切 | 长文档 | 可能切断一天的完整叙事 |
| 按段落切 | 结构清晰文本 | 日记段落长短不一 |
| **先按条目，再按长度** | 日记 | ✅ 保留日期边界 |

### 实现

```python
def chunk_text(
    text: str,
    max_chars: int = 500,
    overlap_chars: int = 50,
) -> list[str]:
    """将长文本切成带重叠的块。"""
    if len(text) <= max_chars:
        return [text]

    chunks = []
    start = 0
    while start < len(text):
        end = start + max_chars
        chunk = text[start:end]

        # 尽量在句号处断开
        if end < len(text):
            last_period = chunk.rfind("。")
            if last_period > max_chars // 2:
                end = start + last_period + 1
                chunk = text[start:end]

        chunks.append(chunk.strip())
        start = end - overlap_chars

    return [c for c in chunks if c]
```

### 生成 Chunk 对象

```python
@dataclass
class Chunk:
    id: str              # "2025-06-15_0"
    date: str
    text: str
    chunk_index: int
    source_file: str
    word_count: int


def entry_to_chunks(entry: DiaryEntry, max_chars: int, overlap: int) -> list[Chunk]:
    texts = chunk_text(entry.content, max_chars, overlap)
    return [
        Chunk(
            id=f"{entry.date}_{i}",
            date=entry.date,
            text=t,
            chunk_index=i,
            source_file=entry.source_file,
            word_count=len(t),
        )
        for i, t in enumerate(texts)
    ]
```

## 4.4 SQLite 存储

### 表结构

```sql
CREATE TABLE IF NOT EXISTS chunks (
    id          TEXT PRIMARY KEY,
    date        TEXT NOT NULL,
    text        TEXT NOT NULL,
    chunk_index INTEGER NOT NULL,
    source_file TEXT,
    word_count  INTEGER,
    created_at  TEXT DEFAULT (datetime('now'))
);

CREATE INDEX IF NOT EXISTS idx_chunks_date ON chunks(date);
```

### store.py 扩展

```python
import sqlite3
from src.store import load_config, resolve_path


def get_db() -> sqlite3.Connection:
    cfg = load_config()
    db_path = resolve_path(cfg["data"]["db_path"])
    db_path.parent.mkdir(parents=True, exist_ok=True)
    conn = sqlite3.connect(db_path)
    conn.row_factory = sqlite3.Row
    _init_tables(conn)
    return conn


def _init_tables(conn: sqlite3.Connection):
    conn.executescript("""
        CREATE TABLE IF NOT EXISTS chunks (
            id TEXT PRIMARY KEY,
            date TEXT NOT NULL,
            text TEXT NOT NULL,
            chunk_index INTEGER NOT NULL,
            source_file TEXT,
            word_count INTEGER,
            created_at TEXT DEFAULT (datetime('now'))
        );
        CREATE INDEX IF NOT EXISTS idx_chunks_date ON chunks(date);
    """)
    conn.commit()


def save_chunks(chunks: list[Chunk], conn: sqlite3.Connection):
    for c in chunks:
        conn.execute(
            """INSERT OR REPLACE INTO chunks
               (id, date, text, chunk_index, source_file, word_count)
               VALUES (?, ?, ?, ?, ?, ?)""",
            (c.id, c.date, c.text, c.chunk_index, c.source_file, c.word_count),
        )
    conn.commit()
```

## 4.5 ingest.py 主流程

```python
"""日记导入：解析 → 切块 → 存库。"""

from pathlib import Path
from src.store import load_config, resolve_path, get_db, save_chunks


def ingest_all() -> int:
    cfg = load_config()
    diary_dir = resolve_path(cfg["data"]["diary_dir"])
    max_chars = cfg["chunking"]["max_chars"]
    overlap = cfg["chunking"]["overlap_chars"]
    date_pattern = cfg["diary"]["date_pattern"]

    all_chunks = []
    for filepath in sorted(diary_dir.glob("*.md")):
        entries = parse_diary_file(str(filepath), date_pattern)
        for entry in entries:
            all_chunks.extend(entry_to_chunks(entry, max_chars, overlap))

    conn = get_db()
    save_chunks(all_chunks, conn)
    conn.close()
    return len(all_chunks)


if __name__ == "__main__":
    count = ingest_all()
    print(f"导入完成，共 {count} 个 chunk")
```

## 4.7 切块参数调优

| 参数 | 太小 | 太大 | 推荐 |
|------|------|------|------|
| max_chars | 检索精准但丢上下文 | 语义模糊 | 400–500 |
| overlap_chars | 边界信息丢失 | 冗余存储 | 50–80 |

10 万字、500 字/块 → 约 200 个 chunk，对 SQLite 和 Chroma 都很轻松。
